        if path == "/api/wv/project-evidence":
            qs = urllib.parse.parse_qs(parsed.query)
            got = wv_project_evidence(qs.get("project", [""])[0], qs.get("ticket", [""])[0], qs.get("name", [""])[0])
            if not got:
                self.send_error(404, "Evidence file not found")
                return
            self.send_response(200)
            self.send_header("Content-Type", got[0])
            self.send_header("Content-Length", str(len(got[1])))
            self.send_header("X-Content-Type-Options", "nosniff")
            self.send_header("Content-Security-Policy", EVIDENCE_CSP)
            self.send_header("Cache-Control", "no-store")
            self.end_headers()
            self.wfile.write(got[1])
            return
