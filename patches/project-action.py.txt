def wv_project_ticket_dir(root, tid):
    base = Path(root) / "specs" / "tickets"
    try:
        for d in sorted(base.iterdir()):
            if d.is_dir() and d.name.startswith(tid + "-"):
                return d
    except OSError:
        pass
    return None


def wv_project_set_status(root, tid, new_status):
    """Board row and rd.md header of a ticket of another project: the same two edits the dashboard makes for its own."""
    board = Path(root) / "specs" / "tickets" / "BOARD.md"
    text = board.read_text(encoding="utf-8")
    new_text, n = re.subn(rf"^(\|\s*{tid}\s*\|[^|]*\|\s*)[a-z-]+(\s*\|)", lambda m: m.group(1) + new_status + "   " + m.group(2), text, count=1, flags=re.M)
    if not n:
        return False
    board.write_text(new_text, encoding="utf-8", newline="")
    d = wv_project_ticket_dir(root, tid)
    if d and (d / "rd.md").is_file():
        rd = (d / "rd.md").read_text(encoding="utf-8")
        rd2 = re.sub(r"^(\*\*Status:\*\*\s*).*$", lambda m: m.group(1) + new_status, rd, count=1, flags=re.M)
        (d / "rd.md").write_text(rd2, encoding="utf-8", newline="")
    return True


def wv_project_action(req):
    """One ticket action on any registered project from the one board: dispatch (draft to ready) or pause. Done here, on
    that project's own files: no second dashboard is involved."""
    name, tid, op = str(req.get("project") or ""), str(req.get("ticket") or ""), str(req.get("op") or "")
    fleet = next((f for f in wv_fleets() if f["name"] == name), None)
    if not fleet or not re.fullmatch(r"\d{3}", tid) or op not in ("dispatch", "pause"):
        return 400, {"error": "Unknown project, ticket or operation."}
    root = fleet["root"]
    row = next((r for r in wv_fleet_rows(root) if r["id"] == tid), None)
    if not row:
        return 404, {"error": f"Ticket {tid} not found in {name}."}
    if op == "dispatch":
        if row["status"] != "draft":
            return 409, {"error": f"Ticket {tid} is '{row['status']}', not a draft."}
        wv_project_set_status(root, tid, "ready")
        return 200, {"ok": True, "note": f"Ticket {tid} of {name} is ready; its router picks it up on the next cycle."}
    if row["status"] in ("done", "failed", "draft"):
        return 409, {"error": f"Ticket {tid} is '{row['status']}', nothing to pause."}
    ctl = Path(root) / "job" / "control"
    ctl.mkdir(parents=True, exist_ok=True)
    (ctl / f"{tid}.json").write_text(json.dumps({"action": "pause_ticket", "issuedAt": utc_now_iso(), "by": "dashboard"}) + "\n", encoding="utf-8")
    return 200, {"ok": True, "note": f"Pause queued for ticket {tid} of {name}."}


def wv_project_ticket(project, tid):
    """A ticket of another project in exactly the shape /api/ticket/<id> has, so the one dashboard's drawer shows it."""
    fleet = next((f for f in wv_fleets() if f["name"] == project), None)
    if not fleet or not re.fullmatch(r"\d{3}", str(tid)):
        return 404, {"error": "Unknown project or ticket."}
    row = next((r for r in wv_fleet_rows(fleet["root"]) if r["id"] == tid), None)
    d = wv_project_ticket_dir(fleet["root"], tid)
    if not d:
        return 404, {"error": f"Ticket {tid} not found in {project}."}

    def rd_file(name):
        p = d / name
        return p.read_text(encoding="utf-8", errors="ignore") if p.is_file() else None

    results_json = None
    try:
        if (d / "results.json").is_file():
            parsed = json.loads((d / "results.json").read_text(encoding="utf-8"))
            results_json = parsed if isinstance(parsed, dict) else None
    except (OSError, ValueError):
        results_json = None
    evidence = []
    if (d / "evidence").is_dir():
        for ef in sorted((d / "evidence").iterdir()):
            if ef.is_file():
                evidence.append({"name": ef.name, "isImage": ef.suffix.lower() in EVIDENCE_IMAGE_TYPES, "size": ef.stat().st_size,
                                 "url": "/api/wv/project-evidence?project=" + urllib.parse.quote(project) + "&ticket=" + tid + "&name=" + urllib.parse.quote(ef.name)})
    return 200, {"id": tid, "project": project, "title": (row or {}).get("title", ""), "status": (row or {}).get("status", ""),
                 "folder": d.name, "rd": rd_file("rd.md"), "eval": rd_file("eval.md"), "contract": rd_file("contract.md"),
                 "results": rd_file("results.md"), "resultsJson": results_json, "decision": rd_file("user-decision.md"),
                 "mockup": False, "evidence": evidence, "git": None}


def wv_project_evidence(project, tid, name):
    fleet = next((f for f in wv_fleets() if f["name"] == project), None)
    d = wv_project_ticket_dir(fleet["root"], tid) if fleet and re.fullmatch(r"\d{3}", str(tid)) else None
    if not d or "/" in name or chr(92) in name or name.startswith("."):
        return None
    f = d / "evidence" / name
    if not f.is_file() or not f.resolve().is_relative_to((d / "evidence").resolve()):
        return None
    mime = EVIDENCE_IMAGE_TYPES.get(f.suffix.lower(), EVIDENCE_TEXT_TYPE)
    data = f.read_bytes()
    if mime == EVIDENCE_TEXT_TYPE:
        data = scrub_secret_bytes(data, live_secret_values())
    return mime, data
