def wv_project_action(req):
    """One ticket action on any registered project, from the one board: dispatch (draft to ready), pause or restart. The
    action is forwarded to that project's own dashboard with its own token, which never reaches the browser."""
    import urllib.request
    import urllib.error
    name, tid, op = str(req.get("project") or ""), str(req.get("ticket") or ""), str(req.get("op") or "")
    fleet = next((f for f in wv_fleets() if f["name"] == name), None)
    body = {
        "dispatch": {"action": "set_ticket_status", "ticket": tid, "status": "ready"},
        "pause": {"action": "pause_ticket", "ticket": tid},
        "restart": {"action": "wv_retry_ticket", "ticket": tid},
    }.get(op)
    if not fleet or not re.fullmatch(r"\d{3}", tid) or body is None:
        return 400, {"error": "Unknown project, ticket or operation."}
    root = fleet["root"]
    port = ((wv_fleet_json(root, "specs/worker-fleet.config.json").get("dashboard") or {}).get("port")) or 3080
    token = wv_fleet_json(root, "job/dashboard-auth.json").get("authToken")
    if not token or not wv_port_open(port):
        return 409, {"error": f"{name}'s dashboard is not running yet. It is started automatically; try again in a minute."}
    req2 = urllib.request.Request(f"http://127.0.0.1:{port}/api/action", data=json.dumps(body).encode("utf-8"), method="POST")
    req2.add_header("content-type", "application/json")
    req2.add_header("authorization", "Bearer " + token)
    req2.add_header("origin", f"http://127.0.0.1:{port}")
    try:
        with urllib.request.urlopen(req2, timeout=30) as r:
            return r.status, json.loads(r.read().decode("utf-8") or "{}")
    except urllib.error.HTTPError as e:
        try:
            return e.code, json.loads(e.read().decode("utf-8") or "{}")
        except Exception:
            return e.code, {"error": f"{name} answered {e.code}"}
    except Exception as e:
        return 502, {"error": str(e)[:160]}
