WV_FLEETS_FILE = Path.home() / ".workerov" / "fleets.json"


def wv_norm_root(p):
    return str(p).replace("\\", "/").rstrip("/").lower()


def wv_fleets():
    """Every fleet registered on this machine ([{name, root}] in ~/.workerov/fleets.json); this one is always listed."""
    try:
        items = json.loads(WV_FLEETS_FILE.read_text(encoding="utf-8"))
    except Exception:
        items = []
    out, seen = [], set()
    for it in (items if isinstance(items, list) else []) + [{"name": REPO_ROOT.name, "root": str(REPO_ROOT)}]:
        if not isinstance(it, dict):
            continue
        root = str(it.get("root", "")).replace("\\", "/").rstrip("/")
        key = wv_norm_root(root)
        if not root or key in seen:
            continue
        seen.add(key)
        out.append({"name": str(it.get("name") or Path(root).name), "root": root})
    return out


def wv_port_open(port):
    import socket
    try:
        with socket.create_connection(("127.0.0.1", int(port)), 0.4):
            return True
    except (OSError, ValueError, TypeError):
        return False


def wv_fleet_json(root, rel):
    try:
        return json.loads((Path(root) / rel).read_text(encoding="utf-8"))
    except Exception:
        return {}


def wv_fleet_rows(root):
    rows = []
    try:
        text = (Path(root) / "specs" / "tickets" / "BOARD.md").read_text(encoding="utf-8")
    except OSError:
        return rows
    for m in re.finditer(r"^\|\s*(\d{3})\s*\|\s*([^|]*?)\s*\|\s*([a-z-]+)\s*\|\s*([a-z]+)?", text, re.M):
        rows.append({"id": m.group(1), "title": m.group(2), "status": m.group(3), "mode": m.group(4) or ""})
    return rows


def wv_projects():
    """One board for every project on this computer: each fleet's tickets plus which worker is alive (one at a time)."""
    projects = []
    for f in wv_fleets():
        root = f["root"]
        cfg = wv_fleet_json(root, "specs/worker-fleet.config.json")
        port = ((cfg.get("dashboard") or {}).get("port")) or 3080
        pending = {str(x).split("-")[0] for x in (wv_fleet_json(root, "job/router-state.json").get("pendingDecisions") or [])}
        live = {}
        for e in wv_fleet_json(root, "job/router-state.json").get("dispatched", []) or []:
            if isinstance(e, dict) and wv_port_open(e.get("port")):
                live[str(e.get("ticket", "")).split("-")[0]] = e.get("port")
        rows = wv_fleet_rows(root)
        for r in rows:
            r["running"] = r["id"] in live
            r["awaiting"] = r["status"] != "done" and r["id"] in pending
        turn = ""
        try:
            tp = Path.home() / ".wv-bin" / "wv-turn.py"
            if tp.is_file():
                r = subprocess.run([sys.executable, "-I", str(tp), root, "--peek"], capture_output=True, text=True, timeout=8, creationflags=WV_NOWIN)
                turn = (r.stdout or "").strip()
        except Exception:
            turn = ""
        cur = wv_router_project()
        if cur and wv_norm_root(cur.get("root", "")) == wv_norm_root(root):
            turn = "the router is working on this project now"
        projects.append({
            "router": turn,
            "name": f["name"], "root": root, "port": port, "online": wv_port_open(port),
            "off": (Path(root) / "job" / "fleet-off").exists(), "tickets": rows, "runningCount": len(live),
            "self": wv_norm_root(root) == wv_norm_root(REPO_ROOT),
        })
    return {"projects": projects, "workerActive": [p["name"] for p in projects if p["runningCount"]]}


def wv_wanted():
    """The one router serves every project: it is wanted when any registered project needs it (wv-turn.py --pick)."""
    try:
        r = subprocess.run([sys.executable, "-I", str(Path.home() / ".wv-bin" / "wv-turn.py"), "--pick", "--peek"],
                           capture_output=True, text=True, timeout=20)
        return r.returncode == 0
    except Exception:
        return False


def wv_router_project():
    """The project the one router is working on right now (the permission plugin's guard file), or None."""
    try:
        d = json.loads((Path.home() / ".workerov" / "router-project.json").read_text(encoding="utf-8"))
        return d if d.get("name") else None
    except Exception:
        return None
